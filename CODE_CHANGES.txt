@staticmethod
def remove_characters(text: str) -> str:
    """Remove non-letter characters from a given string."""
    if text is None:  # <- Now handles None
        return ""
    text = str(text)
    text = re.sub(r'https?://\S+', '', text)  # <- Removes URLs
    return re.sub(r'[^A-Za-z]+', '', text)  # <- Removes all non-letters

DEFAULT_API_URLS = [
    os.getenv("PREDICTION_API_URL"),
    "http://localhost:8000/get-prediction/",
    "http://model_inference_endpoint:8000/get-prediction/",
]

def get_prediction(input_text):
    # TODO - task 3
    # Old placeholder code:
    # pass
    
    if not input_text or not input_text.strip():
        return "No tweet text provided."

    payload = {"input_texts": input_text}

    for api_url in [url for url in DEFAULT_API_URLS if url]:
        try:
            response = requests.post(api_url, json=payload, timeout=5)
            response.raise_for_status()
            prediction_data = response.json()
            return prediction_data.get("prediction", "No prediction returned")
        except requests.RequestException:
            continue

    return (
        "Unable to reach the prediction service. "
        "Set PREDICTION_API_URL or run the model inference endpoint first."
    )

================================================================================
HOW THE SYSTEM WORKS (ARCHITECTURE)
================================================================================

FLOW:
1. User enters tweet in Streamlit app (src/app/app.py)
2. App calls FastAPI endpoint at http://localhost:8000/get-prediction/
3. Endpoint receives tweet (src/model-inference-endpoint/main.py)
4. Endpoint processes tweet using trained model:
   - Cleans text (removes URLs, punctuation)
   - Scores tokens against training vocabulary
   - Returns prediction: "Republican" or "Democrat"
5. App displays prediction to user

DATA FLOW:
training data (data/Tweets.csv)
      ↓
DataLoader (src/text_loader/loader.py)
      ↓
Vectorizer & Label Encoder
      ↓
Model Inference Endpoint (src/model-inference-endpoint/main.py)
      ↓
Streamlit App (src/app/app.py)
      ↓
Web UI (http://localhost:8501)


================================================================================
KEY DESIGN DECISIONS
================================================================================

1. KEYWORD-BASED PREDICTION
   - Reason: Simple, interpretable, no neural network required
   - Data-driven: Uses actual token frequencies from training data
   - Fast: O(1) per token lookup

2. FALLBACK TO MLFLOW MODEL
   - Reason: Allows upgrading to trained model without code change
   - Path: Look for model in ./data/model/ or ../data/model/
   - Graceful: Falls back to keyword scoring if model not found

3. MULTIPLE API URL SUPPORT
   - Development: http://localhost:8000
   - Docker Compose: http://model_inference_endpoint:8000
   - Environment: Custom PREDICTION_API_URL env variable

4. INPUT VALIDATION
   - Checks for None, empty strings
   - Handles non-string inputs (converts to string)
   - Removes URLs before processing

5. ERROR HANDLING
   - Network errors: Tries fallback URLs
   - Missing data: Raises FileNotFoundError with clear message
   - Invalid input: Returns user-friendly message
